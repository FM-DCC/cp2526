$$CM:= \text{coin.coffee}.CM$$

\part \pten{Escreve um processo  que funcione como}{Write a process that behaves as} $CM$ \pten{mas  pode dar chá ou café pelo mesmo preço}{but can give coffee or tea}.\pten{Considera os casos em que a bebida que sai é aleatória ou que o comprador indica se quer café ou chá}{Consider two cases: the beverage is randomlly choosen or there is a way to ask the buyer if he wants coffee or tea}.
\part \pten{Escreve um processo  que funcione como}{Write a process that behaves as} $CM$ \pten{mas pode dar chá  por $0.5$ euros}{but can give tea for $0.5$ euros} \pten{e}{and} \pten{café  por}{ coffee for} $1$ euro.
\part \pten{Para as últimas alíneas repete a primeira alínea}{Repeat} (\ref{ex:cma}) \pten{}{for the last questions}.
\end{parts}
